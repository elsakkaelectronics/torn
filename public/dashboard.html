app.get('/api/bars', authenticateApiKey, async (req, res) => { const user = await get('SELECT api_key FROM users WHERE id = ?', req.user.id); if (!user || !user.api_key) { return res.status(400).json({ error: 'Torn API key not set' }); } try { const response = await fetch(`https://api.torn.com/v2/user/bars?key=${user.api_key}`); if (!response.ok) throw new Error('Failed to fetch bars'); const data = await response.json(); if (data.error) throw new Error(data.error); res.json(data.bars || {}); } catch (e) { res.status(500).json({ error: e.message }); } });
